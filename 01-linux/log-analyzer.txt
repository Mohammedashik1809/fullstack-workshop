# Check if the log file exists
if [ ! -f "$LOG_FILE" ]; then
    echo "Error: Log file '$LOG_FILE' not found!"
    exit 1
fi

# Count total number of lines in the log file
TOTAL_LINES=$(wc -l < "$LOG_FILE")

# Count occurrences of log levels
INFO_COUNT=$(grep -c "INFO" "$LOG_FILE" || true)
WARNING_COUNT=$(grep -c "WARNING" "$LOG_FILE" || true)
ERROR_COUNT=$(grep -c "ERROR" "$LOG_FILE" || true)

# Extract unique IP addresses from the log file
IP_LIST=$(grep -oE '[0-9]{1,3}(\.[0-9]{1,3}){3}' "$LOG_FILE" | sort -u || true)

# Display analysis report
echo "========== LOG ANALYSIS REPORT =========="
echo "File: $LOG_FILE"
echo "Total Lines: $TOTAL_LINES"
echo "----------------------------------------"
echo "INFO:    $INFO_COUNT"
echo "WARNING: $WARNING_COUNT"
echo "ERROR:   $ERROR_COUNT"
echo "----------------------------------------"

# Display unique IP addresses if found
if [ -z "$IP_LIST" ]; then
    echo "Unique IP Addresses Found: None"
else
    echo "Unique IP Addresses Found:"
    while read -r ip; do
        echo " - $ip"
    done <<< "$IP_LIST"
fi

echo "========================================"
